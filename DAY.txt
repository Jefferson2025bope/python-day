class SuperHero:
    def __init__(self, name="", alias="", special_skill=""):
        self.name = name
        self.alias = alias
        self.special_skill = special_skill

    def __str__(self):
        return f"{self.name}: {self.alias} - {self.special_skill}"


# Criar super-heróis
super_hero1 = SuperHero()
super_hero1.name = "Peter Parker"
super_hero1.alias = "Spiderman"
super_hero1.special_skill = "Web Slinging"

super_hero2 = SuperHero("Tony Stark", "Iron Man", "Flying")

super_hero3 = SuperHero(
    name="Bruce Banner",
    alias="The Hulk",
    special_skill="Strength"
)

# Imprimir super-heróis individuais
print(super_hero1)
print(super_hero2)
print(super_hero3)

# Imprimir usando f-strings
print(f"Hero: {super_hero1.name}:{super_hero1.alias}")
print(f"Hero: {super_hero2.name}:{super_hero2.alias}")
print(f"Hero: {super_hero3.name}:{super_hero3.alias}")

# Lista de heróis da DC (usando underscore '_' em vez de hífen)
dc_heros = [
    SuperHero("Bruce Wayne", "Batman", "Technology"),
    SuperHero("Clark Kent", "Superman", "Flying"),
    SuperHero("Diana Prince", "Wonder Woman", "Combat")
]

print("\n--- Lista de Heróis da DC ---")
for hero in dc_heros:
    print(hero)
